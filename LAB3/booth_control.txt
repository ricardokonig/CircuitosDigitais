module booth_control (
    input  wire clk,
    input  wire rst,
    input  wire start,
    input  wire count_zero,
    input  wire q0,
    input  wire q_minus_1,
    output reg  load,
    output reg  do_add,
    output reg  do_sub,
    output reg  do_shift,
    output reg  do_dec,
    output reg  done
);

    localparam [2:0] S_IDLE  = 3'd0,
                     S_INIT  = 3'd1,
                     S_CHECK = 3'd2,
                     S_ADD   = 3'd3,
                     S_SUB   = 3'd4,
                     S_SHIFT = 3'd5,
                     S_DEC   = 3'd6,
                     S_DONE  = 3'd7;

    reg [2:0] state, next_state;

    // Registrador de Estado
    always @(posedge clk or posedge rst) begin
        if (rst)
            state <= S_IDLE;
        else
            state <= next_state;
    end

    // Lógica de Próximo Estado
    always @(*) begin
        next_state = state;
        case (state)
            S_IDLE:  if (start) next_state = S_INIT;
            S_INIT:  next_state = S_CHECK;
            S_CHECK: begin
                if (count_zero)
                    next_state = S_DONE;
                else if ({q0, q_minus_1} == 2'b01)
                    next_state = S_ADD;
                else if ({q0, q_minus_1} == 2'b10)
                    next_state = S_SUB;
                else
                    next_state = S_SHIFT;
            end
            S_ADD:   next_state = S_SHIFT;
            S_SUB:   next_state = S_SHIFT;
            S_SHIFT: next_state = S_DEC;
            S_DEC:   next_state = S_CHECK;
            S_DONE:  if (!start) next_state = S_IDLE;
            default: next_state = S_IDLE;
        endcase
    end

    // Lógica de Saída (Moore)
    always @(*) begin
        load     = 1'b0;
        do_add   = 1'b0;
        do_sub   = 1'b0;
        do_shift = 1'b0;
        do_dec   = 1'b0;
        done     = 1'b0;

        case (state)
            S_INIT:  load     = 1'b1;
            S_ADD:   do_add   = 1'b1;
            S_SUB:   do_sub   = 1'b1;
            S_SHIFT: do_shift = 1'b1;
            S_DEC:   do_dec   = 1'b1;
            S_DONE:  done     = 1'b1;
        endcase
    end

endmodule
